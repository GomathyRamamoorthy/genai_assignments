```java
package org.testleaf.week5.day1;

import java.time.Duration;
import java.util.logging.Level;
import java.util.logging.Logger;

import org.openqa.selenium.By;
import org.openqa.selenium.WebElement;
import org.openqa.selenium.chrome.ChromeDriver;
import org.openqa.selenium.support.ui.ExpectedConditions;
import org.openqa.selenium.support.ui.WebDriverWait;

/**
 * LearnNestedFrame class.
 */
public class LearnNestedFrame {

    private static final Logger LOGGER = Logger.getLogger(LearnNestedFrame.class.getName());

    /**
     * Main method.
     * 
     * @param args command-line arguments
     */
    public static void main(String[] args) {
        try {
            // Launch the browser
            ChromeDriver driver = new ChromeDriver();

            // Load the URL
            driver.get("https://leafground.com/frame.xhtml");

            // Maximize the browser
            driver.manage().window().maximize();

            // Add implicit wait
            driver.manage().timeouts().implicitlyWait(Duration.ofSeconds(15));

            // Switch inside the outer frame
            WebElement element = driver.findElement(By.xpath("//h5[text()=' Click Me (Inside Nested frame)']/following-sibling::iframe"));
            driver.switchTo().frame(element);

            // Switch inside the inner frame
            driver.switchTo().frame(0);

            // Click the button
            WebDriverWait wait = new WebDriverWait(driver, Duration.ofSeconds(10));
            wait.until(ExpectedConditions.elementToBeClickable(By.id("Click")));
            driver.findElement(By.id("Click")).click();

            // Switch to the main page
            driver.switchTo().defaultContent();

            // Switch back to the immediate parent frame
            driver.switchTo().parentFrame();

            // Close the browser
            driver.quit();
        } catch (Exception e) {
            LOGGER.log(Level.SEVERE, "Error occurred: " + e.getMessage());
        }
    }
}
```

Note that the refactored code includes try-catch blocks, JavaDoc comments, and logging. Additionally, it uses more efficient locators and reduces the number of `findElement` calls. However, it still lacks unit tests and input validation, which should be added to improve the code's maintainability and robustness.
